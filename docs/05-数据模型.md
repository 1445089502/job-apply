# 05 · 数据模型（表结构）

## 5.1 核心表一览
| 表 | 用途 |
|---|---|
| `company` | 公司信息 |
| `job` | 岗位信息（含向量） |
| `job_application` | 投递记录（状态机） |
| `resume` | 简历版本 |
| `match_score` | 匹配评分明细 |
| `apply_event` | 投递流水（审计） |
| `daily_report` | 日报预聚合 |

## 5.2 表结构设计

### company
```sql
CREATE TABLE company (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name        VARCHAR(128) NOT NULL,
  industry    VARCHAR(64),
  city        VARCHAR(32),
  scale       VARCHAR(32),              -- 规模
  url         VARCHAR(255),
  created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_name (name)
) ENGINE=InnoDB;
```

### job
```sql
CREATE TABLE job (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  platform      VARCHAR(16) NOT NULL,      -- boss/zhaopin/...
  job_id        VARCHAR(64) NOT NULL,      -- 平台岗位ID
  company_id    BIGINT UNSIGNED,
  title         VARCHAR(128) NOT NULL,
  jd_text       TEXT NOT NULL,
  city          VARCHAR(32),
  district      VARCHAR(32),
  salary_min    INT,                        -- 归一为 千/月 或 万/年
  salary_max    INT,
  salary_currency VARCHAR(8) DEFAULT 'CNY',
  experience_req VARCHAR(32),
  education_req  VARCHAR(32),
  language_req   VARCHAR(64),
  skills_json    JSON,
  embedding      BLOB,                      -- bge-m3 向量(或用FAISS另存)
  status         TINYINT DEFAULT 1,         -- 1有效 0失效
  created_at     DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_platform_job (platform, job_id),
  KEY idx_city_salary (city, salary_min),
  KEY idx_status (status)
) ENGINE=InnoDB;
```

### job_application（状态机）
```sql
CREATE TABLE job_application (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id     BIGINT UNSIGNED NOT NULL,
  job_id      BIGINT UNSIGNED NOT NULL,
  platform    VARCHAR(16) NOT NULL,
  status      VARCHAR(16) NOT NULL,        -- CREATED/QUEUED/APPLYING/SUCCESS/FAILED/NEEDS_HUMAN
  version     INT NOT NULL DEFAULT 0,      -- 乐观锁
  resume_id   BIGINT UNSIGNED,
  attempts    INT DEFAULT 0,
  last_error  VARCHAR(255),
  applied_at  DATETIME,
  created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at  DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_user_job (user_id, platform, job_id),
  KEY idx_status (status),
  KEY idx_applied (applied_at)
) ENGINE=InnoDB;
```

### match_score
```sql
CREATE TABLE match_score (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id     BIGINT UNSIGNED NOT NULL,
  job_id      BIGINT UNSIGNED NOT NULL,
  hard_score  DECIMAL(3,2),       -- 硬过滤
  vec_score   DECIMAL(3,2),       -- 向量
  llm_score   DECIMAL(3,2),       -- LLM
  final_score DECIMAL(3,2),
  reasons_json JSON,              -- LLM 理由
  created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_user_job (user_id, job_id)
) ENGINE=InnoDB;
```

### resume
```sql
CREATE TABLE resume (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id     BIGINT UNSIGNED NOT NULL,
  version     INT NOT NULL,
  file_path   VARCHAR(255) NOT NULL,   -- 对象存储路径(加密)
  template    VARCHAR(64),
  created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_user_version (user_id, version)
) ENGINE=InnoDB;
```

### apply_event（审计流水）
```sql
CREATE TABLE apply_event (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  application_id BIGINT UNSIGNED NOT NULL,
  from_status VARCHAR(16),
  to_status   VARCHAR(16),
  detail_json JSON,
  created_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
  KEY idx_app (application_id)
) ENGINE=InnoDB;
```

### daily_report（预聚合）
```sql
CREATE TABLE daily_report (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  report_date DATE NOT NULL,
  platform    VARCHAR(16),
  viewed_cnt  INT,       -- 浏览
  applied_cnt INT,       -- 投递
  read_cnt    INT,       -- 已读
  replied_cnt INT,       -- 回复
  interview_cnt INT,     -- 面试
  offer_cnt   INT,       -- offer
  UNIQUE KEY uk_date_platform (report_date, platform)
) ENGINE=InnoDB;
```
